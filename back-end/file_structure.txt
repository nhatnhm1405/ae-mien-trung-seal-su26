CHAY:
    cd back-end
    pip install -r requirements.txt
    python -m app.ml.train_anomaly_model      # bat buoc lan dau, neu khong se khong phat hien duoc gi
    python -m app.ml.train_irrigation_model

    uvicorn app.main:app --port 8000                    # dev, dung du lieu gia lap
    SIMULATE=false uvicorn app.main:app --port 8000     # thi that, ket noi broker BTC

BIEN MOI TRUONG DIEU KHIEN LUC CHAY:
    SIMULATE=false          -> dung broker that thay vi simulator
    AGENT_ON_ANOMALY=false  -> tat tang LLM (van co anomaly detection) khi het quota/mat mang
    AGENT_COOLDOWN_SECONDS  -> giay toi thieu giua 2 lan goi LLM cho CUNG mot thiet bi (mac dinh 60)
